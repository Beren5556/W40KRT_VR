cmake_minimum_required(VERSION 3.20)
project(RTMaquetaOpenXR LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(DYNAMIC_LOADER OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(BUILD_API_LAYERS OFF CACHE BOOL "" FORCE)
set(BUILD_LOADER ON CACHE BOOL "" FORCE)
set(CMAKE_DISABLE_FIND_PACKAGE_Python3 ON)
add_subdirectory(../third_party/OpenXR-SDK-release-1.1.63 openxr)
add_library(RTMaquetaBridge SHARED Bridge.cpp)
target_compile_definitions(RTMaquetaBridge PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX XR_USE_PLATFORM_WIN32 XR_USE_GRAPHICS_API_D3D11)
target_compile_options(RTMaquetaBridge PRIVATE /W4 /EHsc /utf-8 /Zi)
target_link_options(RTMaquetaBridge PRIVATE /DEBUG:FULL /OPT:REF /OPT:ICF /PDBALTPATH:RTMaquetaBridge.pdb)
target_link_libraries(RTMaquetaBridge PRIVATE OpenXR::openxr_loader d3d11 dxgi d3dcompiler)
add_executable(BridgeSmokeTest ../tests/BridgeSmokeTest.cpp)
target_compile_definitions(BridgeSmokeTest PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(BridgeSmokeTest PRIVATE RTMaquetaBridge d3d11 dxgi)
add_executable(RuntimePolicyTests ../tests/RuntimePolicyTests.cpp)
target_link_libraries(RuntimePolicyTests PRIVATE RTMaquetaBridge)
add_executable(Resolution80Tests ../tests/Resolution80Tests.cpp)
